// Initialize the map centered on Gujarat var map = L.map('map').setView([22.2587, 71.1924], 7); // Add OpenStreetMap tiles L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', { maxZoom: 19, attribution: '© OpenStreetMap contributors' }).addTo(map); // Fetch your custom JSON file fetch('data/states/gujarat.json') .then(response => { if (!response.ok) { throw new Error('Network response was not ok'); } return response.json(); }) .then(data => { console.log("Loaded state:", data.state); // Loop through each district for (let districtName in data.districts) { let district = data.districts[districtName]; // Loop through each taluka inside the district for (let talukaName in district.talukas) { let taluka = district.talukas[talukaName]; // Add a clickable marker for each taluka let marker = L.marker([taluka.lat, taluka.lng]).addTo(map); // Click event to zoom in and show info marker.bindPopup(`Taluka: ${talukaName}
District: ${districtName}
Code: ${taluka.code}`); marker.on('click', function() { map.setView([taluka.lat, taluka.lng], 11); }); } } }) .catch(error => { console.error('Failed to load map data:', error); });
